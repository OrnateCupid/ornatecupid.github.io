-- =============================================================================
-- 0. 基础设置 (Options)
-- =============================================================================
local opt = vim.opt

opt.number = true          -- 显示当前行号
opt.relativenumber = true  -- 显示相对行号 (方便 j/k 跳转)
opt.tabstop = 4            -- Tab 占据 4 个空格
opt.shiftwidth = 4         -- 自动缩进时占据 4 个空格
opt.expandtab = true       -- 将 Tab 转换为空格
opt.autoindent = true      -- 自动缩进
opt.cursorline = true      -- 高亮当前行
opt.termguicolors = true   -- 启用真彩色支持
opt.signcolumn = "yes"     -- 总是显示左侧图标列
opt.scrolloff = 8          -- 光标上下保持 8 行距离

-- =============================================================================
-- 1. 基础环境准备 (Bootstrap lazy.nvim)
-- =============================================================================
local lazypath = vim.fn.stdpath("data") .. "/lazy/lazy.nvim"
if not vim.loop.fs_stat(lazypath) then
  vim.fn.system({
    "git",
    "clone",
    "--filter=blob:none",
    "https://github.com/folke/lazy.nvim.git",
    "--branch=stable",
    lazypath,
  })
end
vim.opt.rtp:prepend(lazypath)

-- =============================================================================
-- 2. 插件管理 (lazy.nvim)
-- =============================================================================
require("lazy").setup({
  -- 补全引擎核心与源
  { "hrsh7th/nvim-cmp" },
  { "hrsh7th/cmp-buffer" },
  { "hrsh7th/cmp-path" },
  { "hrsh7th/cmp-nvim-lsp" },
  
  -- 主题插件
  { "folke/tokyonight.nvim", lazy = false, priority = 1000 },

  -- 代码片段引擎
  { "L3MON4D3/LuaSnip" },
  { "saadparwaiz1/cmp_luasnip" },

  -- LSP 配置基础链
  { "williamboman/mason.nvim" },
  { "williamboman/mason-lspconfig.nvim" },
  { "neovim/nvim-lspconfig" },
  
  -- UI 与体验增强插件
  { "onsails/lspkind.nvim" },         -- 补全菜单小图标
  { "windwp/nvim-autopairs" },        -- 【新增】括号自动匹配
})
vim.cmd[[colorscheme tokyonight]]

-- =============================================================================
-- 3. 括号自动匹配配置 (nvim-autopairs)
-- =============================================================================
local autopairs = require("nvim-autopairs")
autopairs.setup({
  disable_filetype = { "TelescopePrompt", "vim" }, -- 在某些特定窗口禁用
})

-- =============================================================================
-- 4. 补全引擎配置 (nvim-cmp)
-- =============================================================================
local cmp = require("cmp")
local luasnip = require("luasnip")
local lspkind = require("lspkind")

-- 【退格静音逻辑】状态变量与事件监听
local last_was_backspace = false
vim.api.nvim_create_autocmd({ "InsertEnter", "InsertCharPre" }, {
  callback = function()
    last_was_backspace = false
  end,
})

cmp.setup({
  -- 动态启用/禁用逻辑
  enabled = function()
    if last_was_backspace then
      return false
    end
    return true
  end,

  -- 补全菜单原生边框 (完美适配 tokyonight 的背景色)
  window = {
    completion = { border = "rounded" },
    documentation = { border = "rounded" },
  },

  -- 格式化：添加 LSP 图标
  formatting = {
    format = lspkind.cmp_format({
      mode = 'symbol_text',
      maxwidth = 50,
      ellipsis_char = '...',
      menu = {
        buffer = "[Buf]",
        nvim_lsp = "[LSP]",
        luasnip = "[Snip]",
        path = "[Path]",
      }
    })
  },

  snippet = {
    expand = function(args)
      luasnip.lsp_expand(args.body)
    end,
  },

  -- 交互快捷键
  mapping = cmp.mapping.preset.insert({
    ["<CR>"]  = cmp.mapping.confirm({ select = true }), 
    ["<Tab>"] = cmp.mapping.confirm({ select = true }), 
    ["<C-Space>"] = cmp.mapping(function()
      last_was_backspace = false
      cmp.complete()
    end, { "i", "s" }),
    
    -- 退格键拦截
    ["<BS>"] = cmp.mapping(function(fallback)
      last_was_backspace = true
      if cmp.visible() then
        cmp.close()
      end
      fallback()
    end, { "i", "s" }),
  }),

  -- 补全源
  sources = cmp.config.sources({
    { name = "nvim_lsp" }, 
    { name = "luasnip" },  
  }, {
    { name = "buffer" },   
    { name = "path" },     
  }),

  completion = {
    completeopt = "menu,menuone,noinsert,popup",
  },
})

-- 【联动】让 autopairs 和 cmp 配合，在补全函数时自动加括号
local cmp_autopairs = require('nvim-autopairs.completion.cmp')
cmp.event:on('confirm_done', cmp_autopairs.on_confirm_done())

-- =============================================================================
-- 5. LSP 基础设施配置 (Mason) & 原生 LSP 激活
-- =============================================================================
require("mason").setup({
  ui = { border = "rounded" },
})

require("mason-lspconfig").setup({
  ensure_installed = { "pyright" },
})

local capabilities = require("cmp_nvim_lsp").default_capabilities()

-- 【全新 API】1. 直接修改 Neovim 原生的 LSP 配置表
vim.lsp.config.pyright = {
  capabilities = capabilities,
}

-- 【全新 API】2. 激活 Python 语言服务器
vim.lsp.enable("pyright")

-- 【全新 API】3. 官方推荐的快捷键挂载方式 (替代旧的 on_attach)
-- 只有当 LSP 成功附加到当前文件时，才会触发这个事件并绑定快捷键
vim.api.nvim_create_autocmd("LspAttach", {
  desc = "LSP 快捷键绑定",
  callback = function(args)
    local bufnr = args.buf
    
    -- 我们把快捷键配置放在这里，保证只有写代码时才生效
    -- 下一步我们可以在这里填入具体的跳转和提示快捷键
  end,
})
